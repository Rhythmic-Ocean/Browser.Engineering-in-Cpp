55class PercentageEncoding {
56 static bool needs_encoding(char ch) {
57 if ((ch >= 'A' && ch <= 'Z') || (ch >= 'a' && ch <= 'z') ||
58 (ch >= '0' && ch <= '9') || ch == '-' || ch == '.' || ch == '_' ||
59 ch == '~')
60 return false;
61 return true;
62 }
63
64public:
65 static std::string encode(const std::string &str) {
66 std::string buf = "";
67 for (auto ch : str) {
68 if (ch == ' ') {
69 buf += "+";
70 } else if (needs_encoding(ch)) {
71 int rep = static_cast<int>(static_cast<unsigned char>(ch));
72 buf += "%" + std::format("{:02X}", rep);
73 } else
74 buf += ch;
75 }
76 return buf;
77 }
