6// This header-only Random namespace implements a self-seeding Mersenne Twister.
7// Requires C++17 or newer.
8// It can be #included into as many code files as needed (The inline keyword
9// avoids ODR violations) Freely redistributable, courtesy of learncpp.com
10// (https://www.learncpp.com/cpp-tutorial/global-random-numbers-random-h/)
11namespace Random {
12// Returns a seeded Mersenne Twister
13// Note: we'd prefer to return a std::seed_seq (to initialize a std::mt19937),
14// but std::seed can't be copied, so it can't be returned by value. Instead,
15// we'll create a std::mt19937, seed it, and then return the std::mt19937 (which
16// can be copied).
17inline std::mt19937 generate() {
18 std::random_device rd{};
19
20 // Create seed_seq with clock and 7 random numbers from std::random_device
21 std::seed_seq ss{
22 static_cast<std::seed_seq::result_type>(
23 std::chrono::steady_clock::now().time_since_epoch().count()),
24 rd(),
25 rd(),
26 rd(),
27 rd(),
28 rd(),
29 rd(),
30 rd()};
31
32 return std::mt19937{ss};
33}
34
35// Here's our global std::mt19937 object.
36// The inline keyword means we only have one global instance for our whole
37// program.
38inline std::mt19937 mt{generate()}; // generates a seeded std::mt19937 and
39 // copies it into our global object
40
41// Generate a random int between [min, max] (inclusive)
42// * also handles cases where the two arguments have different types but can be
43// converted to int
44inline int get(int min, int max) {
45 return std::uniform_int_distribution{min, max}(mt);
46}

48// The following function templates can be used to generate random numbers in
49// other cases

51// See https://www.learncpp.com/cpp-tutorial/function-template-instantiation/
52// You can ignore these if you don't understand them

54// Generate a random value between [min, max] (inclusive)
55// * min and max must have the same type
56// * return value has same type as min and max
57// * Supported types:
58// * short, int, long, long long
59// * unsigned short, unsigned int, unsigned long, or unsigned long long
60// Sample call: Random::get(1L, 6L); // returns long
61// Sample call: Random::get(1u, 6u); // returns unsigned int
62template <typename T> T get(T min, T max) {
63 return std::uniform_int_distribution<T>{min, max}(mt);
64}
65
66// Generate a random value between [min, max] (inclusive)
67// * min and max can have different types
68// * return type must be explicitly specified as a template argument
69// * min and max will be converted to the return type
70// Sample call: Random::get<std::size_t>(0, 6); // returns std::size_t
71// Sample call: Random::get<std::size_t>(0, 6u); // returns std::size_t
72// Sample call: Random::get<std::int>(0, 6u); // returns int
73template <typename R, typename S, typename T> R get(S min, T max) {
74 return get<R>(static_cast<R>(min), static_cast<R>(max));
75}
76} // namespace Random
